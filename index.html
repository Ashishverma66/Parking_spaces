main.py — check_parking_space() 
def check_parking_space(img_pro):
  space_counter = 0

  for pos in pos_List:
    x, y = pos
    # Crop each parking space ROI
    img_crop = img_pro[y:y+height, x:x+width]
    count = cv2.countNonZero(img_crop)

    if count < 900:       # threshold = free
      color     = (0, 255, 0)
      thickness = 5
      space_counter += 1
    else:               # occupied
      color     = (0, 0, 255)
      thickness = 2

    cv2.rectangle(img, pos,
      (pos[0]+width, pos[1]+height),
      color, thickness)

  cvzone.putTextRect(img,
    f'Free: {space_counter}/{len(pos_List)}',
    (100, 50), scale=3, thickness=5)
